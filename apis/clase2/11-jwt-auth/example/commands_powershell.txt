# Exercise 11: JWT Authentication Testing Commands (PowerShell)
# These commands demonstrate the JWT flow with automatic token handling:
# access token (15 min) + refresh token

# 0. (Optional) Put a fixed signing secret in .env (copy .env.example .env) BEFORE "fastapi dev app.py",
#    otherwise a random one is used and tokens die when the server restarts
# python -c "import secrets; print(secrets.token_hex(32))"   -> paste it after JWT_SECRET_KEY= in .env

# 1. Set base URL and headers
$baseUrl = "http://127.0.0.1:8000"
$headers = @{
    "Content-Type" = "application/json"
}

# Helper: show the HTTP status code and body of a failed request
function Show-HttpError($err, $label) {
    $status = [int]$err.Exception.Response.StatusCode
    Write-Host "$label -> HTTP $status $($err.ErrorDetails.Message)" -ForegroundColor Yellow
}

# Helper: decode a JWT payload LOCALLY (never paste real tokens into websites)
function ConvertFrom-JwtPayload($jwt) {
    $payload = $jwt.Split('.')[1].Replace('-', '+').Replace('_', '/')
    switch ($payload.Length % 4) { 2 { $payload += '==' } 3 { $payload += '=' } }
    return [System.Text.Encoding]::UTF8.GetString([System.Convert]::FromBase64String($payload)) | ConvertFrom-Json
}

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "Exercise 11: JWT Authentication Tests" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

# 2. Register a new user
Write-Host "==> Registering user..." -ForegroundColor Cyan
$registerBody = @{
    username = "alice"
    password = "secret123"
} | ConvertTo-Json

try {
    $registerResponse = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/register" -Headers $headers -Body $registerBody
    Write-Host "Registration successful: $($registerResponse.message)" -ForegroundColor Green
    Write-Host ""
} catch {
    Show-HttpError $_ "Registration error (user may already exist)"
    Write-Host ""
}

# 3. Login to get JWT tokens (credentials in JSON body, NOT Basic Auth)
Write-Host "==> Logging in to get access + refresh tokens..." -ForegroundColor Cyan
$loginBody = @{
    username = "alice"
    password = "secret123"
} | ConvertTo-Json

try {
    $loginResponse = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/login" -Headers $headers -Body $loginBody
    $token = $loginResponse.access_token
    $refreshToken = $loginResponse.refresh_token
    Write-Host "Login successful!" -ForegroundColor Green
    Write-Host "Access token:  $($token.Substring(0, 40))..." -ForegroundColor White
    Write-Host "Refresh token: $($refreshToken.Substring(0, 40))..." -ForegroundColor White
    Write-Host "Expires in: $($loginResponse.expires_in) seconds" -ForegroundColor White
    Write-Host ""
} catch {
    Show-HttpError $_ "Login error"
    exit
}

# 4. Inspect the access token payload locally
Write-Host "==> Decoding the access token payload locally..." -ForegroundColor Cyan
$claims = ConvertFrom-JwtPayload $token
$claims | ConvertTo-Json | Write-Host
Write-Host "Lifetime (exp - iat): $($claims.exp - $claims.iat) seconds | iss: $($claims.iss) | aud: $($claims.aud)" -ForegroundColor Green
Write-Host ""

# 5. Set JWT authorization header for protected endpoints
$jwtHeaders = @{
    "Authorization" = "Bearer $token"
}
$refreshHeaders = @{
    "Authorization" = "Bearer $refreshToken"
}

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "PROTECTED ENDPOINTS (access token)" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

# 6. Access user profile with JWT token
Write-Host "==> Getting user profile with JWT token..." -ForegroundColor Cyan
try {
    $profileResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/profile" -Headers $jwtHeaders
    Write-Host "Profile data:" -ForegroundColor Green
    $profileResponse | ConvertTo-Json | Write-Host
    Write-Host ""
} catch {
    Show-HttpError $_ "Profile error"
}

# 7. Get all users with JWT token
Write-Host "==> Getting all users with JWT token..." -ForegroundColor Cyan
try {
    $usersResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers $jwtHeaders
    Write-Host "Users data:" -ForegroundColor Green
    $usersResponse | ConvertTo-Json | Write-Host
    Write-Host ""
} catch {
    Show-HttpError $_ "Users error"
}

# 8. Access protected resource with JWT token
Write-Host "==> Accessing protected resource..." -ForegroundColor Cyan
try {
    $protectedResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/protected" -Headers $jwtHeaders
    Write-Host "Protected resource data:" -ForegroundColor Green
    $protectedResponse | ConvertTo-Json | Write-Host
    Write-Host ""
} catch {
    Show-HttpError $_ "Protected error"
}

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "REFRESH AND LOGOUT" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

# 9. Get a new access token with the REFRESH token
Write-Host "==> Refreshing the access token (with the refresh token)..." -ForegroundColor Cyan
try {
    $refreshResponse = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/refresh" -Headers $refreshHeaders
    $token = $refreshResponse.access_token
    $jwtHeaders = @{ "Authorization" = "Bearer $token" }
    Write-Host "New access token: $($token.Substring(0, 40))..." -ForegroundColor Green
    Write-Host ""
} catch {
    Show-HttpError $_ "Refresh error"
}

# 10. Try /refresh with the ACCESS token (should fail)
Write-Host "==> Refreshing with the ACCESS token (should fail)..." -ForegroundColor Cyan
try {
    Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/refresh" -Headers $jwtHeaders | Out-Null
    Write-Host "Unexpected success!" -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - wrong token type"
    Write-Host ""
}

# 11. Logout: revoke the access token, then the refresh token
Write-Host "==> Logging out (revoking access and refresh tokens)..." -ForegroundColor Cyan
try {
    $logoutAccess = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/logout" -Headers $jwtHeaders
    Write-Host "$($logoutAccess.message)" -ForegroundColor Green
    $logoutRefresh = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/logout" -Headers $refreshHeaders
    Write-Host "$($logoutRefresh.message)" -ForegroundColor Green
    Write-Host ""
} catch {
    Show-HttpError $_ "Logout error"
}

# 12. Use the revoked access token (should fail)
Write-Host "==> Using the revoked access token (should fail)..." -ForegroundColor Cyan
try {
    Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/profile" -Headers $jwtHeaders | Out-Null
    Write-Host "Unexpected success!" -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - token revoked"
    Write-Host ""
}

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "ERROR TESTING" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

# 13. Try to access profile WITHOUT token (should fail)
Write-Host "==> Testing access WITHOUT token (should fail)..." -ForegroundColor Cyan
try {

    Write-Host "Unexpected success!" -ForegroundColor Red
} catch {

    Write-Host ""
}

# 14. Try to login with invalid credentials (should fail)
Write-Host "==> Testing login with invalid credentials (should fail)..." -ForegroundColor Cyan
foreach ($user in @("alice", "nobody")) {
    $invalidLoginBody = @{
        username = $user
        password = "wrongpassword"
    } | ConvertTo-Json

    $timer = [System.Diagnostics.Stopwatch]::StartNew()
    try {
        $invalidLogin = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/login" -Headers $headers -Body $invalidLoginBody
        Write-Host "Unexpected success!" -ForegroundColor Red
    } catch {
        $timer.Stop()
        Show-HttpError $_ "Expected error - Invalid credentials for '$user' ($($timer.ElapsedMilliseconds) ms)"
    }
}
Write-Host "Same answer and similar time: an attacker cannot tell which users exist" -ForegroundColor Green
Write-Host ""

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "All tests completed!" -ForegroundColor Green
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""
Write-Host "Summary:" -ForegroundColor Cyan
Write-Host "- Registration: Creates user account (public endpoint)"
Write-Host "- Login: Sends credentials in JSON body, receives access + refresh tokens"
Write-Host "- Access token: 15 minutes, sent as 'Authorization: Bearer <token>'"
Write-Host "- Refresh token: only for POST /refresh, to get a new access token"
Write-Host "- Logout: revokes the token you send (in-memory blocklist)"
Write-Host "- Every token is checked for signature, exp, iss and aud"
Write-Host "- NO Basic Auth used after login - only JWT tokens!"
